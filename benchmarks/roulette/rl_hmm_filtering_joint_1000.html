The scaling programs of the Roulette paper (Moy et al., PLDI 2025) on ladders of size, and its nine Bayesian networks. A process time is the whole process's, with the language's start and compilation; the other time of the same system leaves them out, as its paper times it. A time is the median of the repetitions below, a failed one counted as infinite. A system run in several forms (ProbLog's knowledge compilers, Ace's routes) has the time of its fastest form.

(require math/bigfloat)
(bf-precision 64)

(require memo
         rackunit
         json)

;; start : [0,1]
;; step : bool -> [0,1]
;; obs : bool -> [0,1]
(define-struct hmm-parameters (start step obs) #:transparent)

;; n-steps : nat
;; parameters : hmm-parameters
;; query : hmm-query(n-steps)
(define-struct configuration (n-steps parameters query) #:transparent)

;; variable(n-steps) = hidden {0..n-steps} + visible {1..n-steps}
(define-struct hidden (id) #:transparent)
(define-struct visible (id) #:transparent)

;; event(n-steps) = variable(n-steps) -assoclist-> bool
;; outcomes(n-steps) : event(n-steps)
;; conditions(n-steps) : event(n-steps)
;; NOTE: assoclist, and not hash, ensures iteration happens in ascending order
;; Random access is never used
(define-struct hmm-query (outcomes conditions) #:transparent)

(define current-hmm-printing (make-parameter #f))

;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
;; Query patterns

;; n-steps:nat -> event(n-steps)
(define (all-visible-variables-true n-steps)
  (for/list ([i (in-range n-steps)]) (cons (visible (add1 i)) #t)))
(define (last-hidden-variable-true n-steps) (list (cons (hidden n-steps) #t)))
(define (first-hidden-variable-true n-steps) (list (cons (hidden 0) #t)))
(define (first-half-hidden-variables-true n-steps)
  (for/list ([i (in-range 0 (floor (/ n-steps 2)))])
    (cons (hidden i) #t)))
(define (second-half-hidden-variables-true n-steps)
  (for/list ([i (in-range (floor (/ n-steps 2)) (add1 n-steps))])
    (cons (hidden i) #t)))

;; n-steps:nat -> hmm-query(n-steps)
(define (smoothing-marginal n-steps)
  (hmm-query (first-hidden-variable-true n-steps)
             (all-visible-variables-true n-steps)))
(define (smoothing-joint n-steps)
  (hmm-query (first-half-hidden-variables-true n-steps)
             (all-visible-variables-true n-steps)))
(define (filtering-marginal n-steps)
  (hmm-query (last-hidden-variable-true n-steps)
             (all-visible-variables-true n-steps)))
(define (filtering-joint n-steps)
  (hmm-query (second-half-hidden-variables-true n-steps)
             (all-visible-variables-true n-steps)))
(define (marginal-likelihood n-steps)
  (hmm-query (all-visible-variables-true n-steps)
             (list)))

;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
;; Roulette implementation

;; configuration -> float
(define (run-hmm-roulette config)
  ;; roulette-model = (variable -hash-> random bool)
  ;; nat parameters -> roulette-model
  (define (make-model n-steps parameters)
    (match-define (hmm-parameters start step obs) parameters)
    (define initial-state (flip start))
    (for/fold ([state initial-state]
               [model (hash (hidden 0) initial-state)]
               #:result model)
              ([i (in-range n-steps)])
      (define state- (if state (flip (step #t)) (flip (step #f))))
      (define observation (if state- (flip (obs #t)) (flip (obs #f))))
      (define i- (add1 i))
      (define model- (hash-set (hash-set model (hidden i-) state-) (visible i-) observation))
      (values state- model-)))
  ;; roulette-model (cons variable bool) -> random bool
  (define (interpret-literal model xb)
    (define v (hash-ref model (car xb)))
    (if (cdr xb) v (! v)))
  ;; (n-steps:nat) roulette-model hmm-query(n-steps) -> random bool
  (define (query-model n-steps model q)
    (match-define (hmm-query outcomes conditions) q)
    (for ([xb conditions])
      (observe! (interpret-literal model xb)))
    (for/fold ([res #t]) ([xb (reverse outcomes)])
      (&& (interpret-literal model xb) res))
    #;(for/fold ([res #t]) ([xb outcomes])
      (&& res (interpret-literal model xb))))
  (match-define (configuration n-steps params q) config)
  (define output ((with-observe (query (query-model n-steps (make-model n-steps params) q) #f)) #t))
  (when (current-hmm-printing)
    (current-hmm-printing output))
  output)

;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
;; Dice implementation

;; configuration -> dice program of type bool
(define (gen-hmm-dice config)
  ;; (n-steps: nat) hmm-parameters ->
  ;; sequence of dice let-bindings for
  ;;   hidden_i where i ∈ {0, ..., n-steps}
  ;;   visible_i where i ∈ {1, ..., n-steps}
  (define (gen-model n-steps params)
    (match-define (hmm-parameters start step obs) params)
    (define next-state
      (string-append
       "fun nextState(prevHidden: bool) {\n"
       (format "  let nextHidden = if prevHidden then flip ~a else flip ~a in\n" (step #t) (step #f))
       (format "  let nextVisible = if nextHidden then flip ~a else flip ~a in\n" (obs #t) (obs #f))
       "  (nextHidden, nextVisible)\n"
       "}\n"))
    (define initializer (format "let hidden0 = flip ~a in\n" start))
    (define stepper
      (apply string-append
             (for/list ([i (in-range n-steps)])
               (define i- (add1 i))
               (string-append
                (format "let state~a = nextState(hidden~a) in\n" i- i)
                (format "let hidden~a = fst state~a in\n" i- i-)
                (format "let visible~a = snd state~a in\n" i- i-)))))
    (string-append next-state initializer stepper))
  ;; ((cons x b): cons variable bool) -> dice code for computing x=b
  (define (gen-literal xb)
    (string-append (if (cdr xb) "" "!")
                   (match (car xb)
                     [(visible i) (format "visible~a" i)]
                     [(hidden i) (format "hidden~a" i)])))
  ;; (n-steps:nat) (q:hmm-query) ->
  ;; dice code for computing q
  ;; given hidden_i (i ∈ 0..n-steps) and visible_j (j ∈ 1..n-steps) are in scope
  (define (gen-query n-steps q)
    (match-define (hmm-query outcomes conditions) q)
    (apply string-append
           (append
            (for/list ([xb conditions])
              (format "let _ = observe(~a) in\n" (gen-literal xb)))
            (list (for/fold ([res "true"]) ([xb outcomes])
                    (format "~a && ~a" res (gen-literal xb)))))))
  (match-define (configuration n-steps params q) config)
  (string-append (gen-model n-steps params) (gen-query n-steps q)))

;; configuration -> float
(define (run-hmm-dice config)
  (define out-string
    (exec DICE-EXE
          (generate-dice (gen-hmm-dice config))
          "-eager" "-show-size" "-num-recursive-calls" "-json"))
  (define results (string->jsexpr out-string))
  (define result (first results))
  (when (current-hmm-printing)
    (current-hmm-printing out-string))
  (string->number
   (hash-ref
    (make-immutable-hash
     (for/list ([kv (in-list (hash-ref result '|Joint Distribution|))])
       (match-define (list k v) kv)
       (cons k v)))
    "true")))

(define (generate-dice content)
  (define out-path (make-temporary-file))
  (with-output-to-file out-path
    #:exists 'replace
    (λ () (displayln content)))
  out-path)

(define DICE-EXE (find-executable-path "dice"))
(define TO 10)

(define (exec . args)
  (define-values (sp out in err)
    (apply subprocess #f #f #f args))
  (sync/timeout TO sp)
  (subprocess-kill sp #t)
  (begin0
    (port->string out)
    (close-input-port out)
    (close-output-port in)
    (close-input-port err)))

;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
;; Dynamic programming implementation

;; hmm-dp-supported-query-type
;;   = all-visible-variables-conditioned
;;   + all-visible-marginal
(define-struct all-visible-variables-conditioned () #:transparent)
(define-struct all-visible-marginal () #:transparent)

;; (n-steps:nat) hmm-query(n-steps) -> hmm-dp-supported-query-type
(define (hmm-dp-query-type n-steps q)
  (match-define (hmm-query outcomes conditions) q)
  (define all-visible-variables (map car (all-visible-variables-true n-steps)))
  (cond
    [(equal? (map car conditions) all-visible-variables) (all-visible-variables-conditioned)]
    [(and (equal? conditions (list)) (equal? (map car outcomes) all-visible-variables)) (all-visible-marginal)]
    [else ('error "hmm-dp-query-type: query not of type supported by DP implementation" n-steps q)]))

(module+ test
  (check-equal? (hmm-dp-query-type 10 (smoothing-marginal 10)) (all-visible-variables-conditioned))
  (check-equal? (hmm-dp-query-type 10 (smoothing-joint 10)) (all-visible-variables-conditioned))
  (check-equal? (hmm-dp-query-type 10 (filtering-marginal 10)) (all-visible-variables-conditioned))
  (check-equal? (hmm-dp-query-type 10 (filtering-joint 10)) (all-visible-variables-conditioned))
  (check-equal? (hmm-dp-query-type 10 (marginal-likelihood 10)) (all-visible-marginal)))

;; {c:configuration | (hmm-dp-query-type c) defined} -> float
(define (run-hmm-dp config)
  (define - bf-)
  (define + bf+)
  (define * bf*)
  (define / bf/)
  (define (lit x) (bf x))
  (define (ber p b) (if b p (- (lit 1) p)))
  (match-define (configuration n-steps (hmm-parameters start step obs) query) config)
  (match-define (hmm-query outcomes conditions) query)
  (define assignments-to-variables (make-immutable-hash (append outcomes conditions)))
  (define (expectation-of f) (for/fold ([acc (lit 0)]) ([x (list #t #f)]) (+ acc (f x))))
  (define/memoize (dp-chunk i hi j hj) ; Pr(visible(i+1) .. visible(j) hidden(j)=hj | hidden(i)=hi)
    (cond
      [(equal? i j) (if (equal? hi hj) (lit 1) (lit 0))] ; Pr(hidden(i)=hj | hidden(i)=hi)
      [else
       (define i+ (add1 i))
       (define vi+ (hash-ref assignments-to-variables (visible i+)))
       ; Pr(vis(i+..j) hid(j)=hj | hid(i)=hi)
       ; = sum{hi+} Pr(hid(i+)=hi+ | hid(i)=hi) Pr(vis(i+..j) hid(j)=hj | hid(i+)=hi+, hid(i)=hi)
       ; = sum{hi+} Pr(hid(i+)=hi+ | hid(i)=hi) Pr(vis(i+..j) hid(j)=hj | hid(i+)=hi+)
       ; = sum{hi+} Pr(hid(i+)=hi+ | hid(i)=hi) Pr(vis(i+)=vi | hid(i)=hi+) Pr(vis(i+2 .. j) hid(j)=hj | hid(i+)=hi+)
       ; = sum{hi+} Pr(hid(i+)=hi+ | hid(i)=hi) Pr(vis(i+)=vi | hid(i)=hi+) (dp-chunk i+ hi+ j hj)
       (expectation-of (λ (hi+)
                 (* (ber (lit (step hi)) hi+)
                    (ber (lit (obs hi+)) vi+)
                    (dp-chunk i+ hi+ j hj))))]))
  (define/memoize (dp-last-chunk i hi j) ; Pr(visible(i+1) .. visible(j) | hidden(i)=hi)
    (expectation-of (λ (hj) (dp-chunk i hi j hj))))
  ; Pr(conditions) is used by both query types
  (define p-conditions
    (expectation-of (λ (h0) (* (ber (lit start) h0) (dp-last-chunk 0 h0 n-steps)))))
  (define result
    (match (hmm-dp-query-type n-steps query)
      [(all-visible-variables-conditioned)
       (define (chunks i hi hiddens)
         (match hiddens
           [(list) (dp-last-chunk i hi n-steps)]
           [(cons (cons (hidden i+) hi+) hiddens+)
            (* (dp-chunk i hi i+ hi+) (chunks i+ hi+ hiddens+))]))
       (define p-conditions-and-outcomes
         (expectation-of (λ (h0) (* (ber (lit start) h0) (chunks 0 h0 outcomes)))))
       (bigfloat->flonum (/ p-conditions-and-outcomes p-conditions))]
      [(all-visible-marginal) p-conditions]))
  (when (current-hmm-printing)
    (current-hmm-printing result))
  result)

;; configuration -> unit
(define (check-hmm config epsilon)
  (define roulette (run-hmm-roulette config))
  (define dice (run-hmm-dice config))
  (define dp (run-hmm-dp config))
  (check-within roulette dice epsilon)
  (check-within dice dp epsilon)
  (check-within roulette dp epsilon))

;; Tests

(module+ test
  (define example-n-steps 80)
  (define epsilon 1e-8)

  ;; Reference implementation of example-configuration
  (define (example-hmm n)
    (define start (flip 0.5))
    (for/fold ([state start]) ([i (in-range n)])
      (define state- (if state (flip 0.3) (flip 0.65)))
      (define observation (if state- (flip 0.65) (flip 0.3)))
      (observe! observation)
      state-))

  ;; Roulette matches the reference implementation
  ;; TODO: shouldn't need to disable optimization (use logprob)
  (check-within
   (run-hmm-roulette (example-configuration example-n-steps))
   ((with-observe (query (example-hmm example-n-steps) #f)) #t)
   epsilon)

  ;; Generated Dice code looks correct
  (check-equal?
   (gen-hmm-dice (example-configuration 3))
   (string-append
    "fun nextState(prevHidden: bool) {\n"
    "  let nextHidden = if prevHidden then flip 0.3 else flip 0.65 in\n"
    "  let nextVisible = if nextHidden then flip 0.65 else flip 0.3 in\n"
    "  (nextHidden, nextVisible)\n"
    "}\n"
    "let hidden0 = flip 0.5 in\n"
    "let state1 = nextState(hidden0) in\n"
    "let hidden1 = fst state1 in\n"
    "let visible1 = snd state1 in\n"
    "let state2 = nextState(hidden1) in\n"
    "let hidden2 = fst state2 in\n"
    "let visible2 = snd state2 in\n"
    "let state3 = nextState(hidden2) in\n"
    "let hidden3 = fst state3 in\n"
    "let visible3 = snd state3 in\n"
    "let _ = observe(visible1) in\n"
    "let _ = observe(visible2) in\n"
    "let _ = observe(visible3) in\n"
    "true && hidden3"))

  ;; Specialized DP for example-configuration
  (define (example-filtering config)
    (match-define (configuration n-steps (hmm-parameters start step obs) _) config)
    ;; (dp sn n) computes P(visible states 1..n all true, hidden state n = sn)
    (define/memoize (dp sn n)
      (cond
        [(equal? n 0) start]
        [else
         (define (prob res p) (if res p (- 1 p)))
         (for/sum ([sn-1 (list #t #f)])
           (* (prob sn (step sn-1)) (obs sn) (dp sn-1 (- n 1))))]))
    (define t (dp #t n-steps))
    (define f (dp #f n-steps))
    (/ t (+ t f)))
  ;; Specialized DP matches reference implementation
  (check-within
   (example-filtering (example-configuration example-n-steps))
   ((with-observe (query (example-hmm example-n-steps) #f)) #t)
   epsilon)

  ;; DP matches specialized DP
  (check-within
   (example-filtering (example-configuration 1))
   (run-hmm-dp (example-configuration 1))
   epsilon)

  ;; Dice, Roulette, DP match
  (define _
    (let ()
      (define (queries n-steps)
        (list (marginal-likelihood n-steps)
              (filtering-marginal n-steps)
              (smoothing-marginal n-steps)
              (filtering-joint n-steps)
              (smoothing-joint n-steps)))
      (for/list ([n-steps (in-range 1 100 10)])
        (for/list ([query (queries n-steps)])
          (define config (configuration n-steps example-hmm-parameters query))
          (check-hmm config epsilon)))))
  )

;; main

(define QUERIES
  (hash 'marginal-likelihood marginal-likelihood
        'filtering-marginal filtering-marginal
        'smoothing-marginal smoothing-marginal
        'filtering-joint filtering-joint
        'smoothing-joint smoothing-joint))

(define RUNNERS
  (hash 'roulette run-hmm-roulette
        'dice run-hmm-dice
        'dp run-hmm-dp))

(module+ main
  (match-define (vector mode query-str step-str)
    (current-command-line-arguments))

  (define query (hash-ref QUERIES (string->symbol query-str)))
  (define n-steps (string->number step-str))
  (define config (configuration n-steps example-hmm-parameters (query n-steps)))
  (define run (hash-ref RUNNERS (string->symbol mode)))
  (parameterize ([current-hmm-printing #t])
    (if (equal? mode "dice")
        (void (run config))
        (time (void (run config))))
    (displayln (current-hmm-printing))))
