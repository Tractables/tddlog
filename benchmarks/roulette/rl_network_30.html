rl_network_30

The scaling programs of the Roulette paper (Moy et al., PLDI 2025) on ladders of size, and its nine Bayesian networks. Roulette runs the artifact's programs; tddlog, Dice and Alea.jl run programs of the same models. On caesar Roulette's time is that of its query with #f, which keeps the answer. 600 s a run. Made by gen_roulette.py network 30.

;; racket network.rkt K: benchmark/scaling/network.rkt of the Roulette
;; artifact under rsdd (`-m rsdd -p (K)`), its distribution printed in
;; full (print.rkt): m(K,V) for a message whose bits, least significant
;; first, make V, m(K,16) for #f (lost). Roulette's distribution also holds #t
;; at a probability of 0 up to rounding; it is printed as other(#t).
(require (submod roulette/benchmark/scaling/network rsdd)
         roulette/example/private/pmf
         racket/format
         "print.rkt")
(define k (string->number (vector-ref (current-command-line-arguments) 0)))
(define result (main k))
(for ([(v p) (in-pmf result)])
  (cond
    [(not v) (emit (format "m(~a,16)" k) p)]
    [(list? v) (emit (format "m(~a,~a)" k (for/sum ([b v] [i (in-naturals)]) (if b (expt 2 i) 0))) p)]
    [else (emit (format "other(~a)" (~s v)) p)]))

fun diamond(m: (bool, (bool, (bool, (bool, bool))))) {
  let lost = fst m in
  let b0 = fst (snd m) in
  let b1 = fst (snd (snd m)) in
  let b2 = fst (snd (snd (snd m))) in
  let b3 = snd (snd (snd (snd m))) in
  let bit = discrete(0.25, 0.25, 0.25, 0.25) in
  let nz = flip 0.01 in
  let fw = flip 0.99 in
  let sp = flip 0.5 in
  let c = nz && !lost in
  let n0 = b0 ^ (c && bit == int(2, 0)) in
  let n1 = b1 ^ (c && bit == int(2, 1)) in
  let n2 = b2 ^ (c && bit == int(2, 2)) in
  let n3 = b3 ^ (c && bit == int(2, 3)) in
  if sp then (lost, (n0, (n1, (n2, n3))))
  else (if fw then m else (true, (false, (false, (false, false)))))
}

using Alea

# below than running them takes: they are interpreted, and Alea's own
# methods compile as usual.
Base.Experimental.@compiler_options compile=min optimize=0 infer=false

const EV = Ref{AnyBool}(true)
observe!(path, c) = (EV[] = EV[] & (!path | c); true)

function shr(x::DistUInt{W}, n::Integer) where W
    k = min(n, W)
    DistUInt{W}(vcat(fill(false, k), x.bits[1:W - k]))
end

components(x::Tuple) = reduce(vcat, map(components, x); init = Any[])
components(x) = Any[x]
width(::DistUInt{W}) where W = W

function warmup()
    a = flip(0.7)
    x = discrete(DistUInt{2}, [0.1, 0.2, 0.3, 0.4])
    y = uniform(DistUInt{3}, 1, 6)
    z = ifelse(a, x + DistUInt{2}(1), x - DistUInt{2}(1))
    t = ifelse(flip(0.4), (a, z), (!a, x * x))
    e = prob_equals(z, DistUInt{2}(2)) | (x < z) | xor(a, flip(0.2))
    pr(flip(0.7))
    pr(a, z, y, t...; evidence = e)
    nothing
end

function d_diamond(x1_path, x2_m)
    x3_lost = x2_m[1]
    x4_b0 = x2_m[2][1]
    x5_b1 = x2_m[2][2][1]
    x6_b2 = x2_m[2][2][2][1]
    x7_b3 = x2_m[2][2][2][2]
    x8_bit = discrete(DistUInt{2}, [0.25, 0.25, 0.25, 0.25])
    x9_nz = flip(0.01)
    x10_fw = flip(0.99)
    x11_sp = flip(0.5)
    x12_c = (x9_nz & !(x3_lost))
    x13_n0 = xor(x4_b0, (x12_c & prob_equals(x8_bit, DistUInt{2}(0))))
    x14_n1 = xor(x5_b1, (x12_c & prob_equals(x8_bit, DistUInt{2}(1))))
    x15_n2 = xor(x6_b2, (x12_c & prob_equals(x8_bit, DistUInt{2}(2))))
    x16_n3 = xor(x7_b3, (x12_c & prob_equals(x8_bit, DistUInt{2}(3))))
    if x11_sp !== false
     x17_diamond = (x3_lost, (x13_n0, (x14_n1, (x15_n2, x16_n3))))
    end
    if x11_sp !== true
     if x10_fw !== true
      x18_diamond = (true, (false, (false, (false, false))))
     end
     x19_diamond = x10_fw === true ? x2_m : x10_fw === false ? x18_diamond : ifelse(x10_fw, x2_m, x18_diamond)
    end
    x20_diamond = x11_sp === true ? x17_diamond : x11_sp === false ? x19_diamond : ifelse(x11_sp, x17_diamond, x19_diamond)
    return x20_diamond
end

function report(k, x, d)
    if x isa AnyBool
        println("alea ", k, " true ", d[true])
        println("alea ", k, " false ", d[false])
    else
        for v in 0:(2^width(x) - 1)
            println("alea ", k, " ", v, " ", d[v])
        end
    end
end

function run_program()
    EV[] = true
    stats = @timed begin
        comps = components(dice_main())
        dists = pr(comps...; evidence = EV[])
        (comps, length(comps) == 1 ? [dists] : dists)
    end
    comps, dists = stats.value
    for k in eachindex(comps)
        report(k, comps[k], dists[k])
    end
    println("alea_seconds ", stats.time)
    println("alea_compile_seconds ", get(stats, :compile_time, NaN))
end

warmup()

schedule(task)
wait(task)
