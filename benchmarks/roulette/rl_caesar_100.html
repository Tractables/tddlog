rl_caesar_100

The scaling programs of the Roulette paper (Moy et al., PLDI 2025) on ladders of size, and its nine Bayesian networks. Roulette runs the artifact's programs; tddlog, Dice and Alea.jl run programs of the same models. On caesar Roulette's time is that of its query with #f, which keeps the answer. 600 s a run. Storm, PRISM, Ace, Merlin, pyAgrum, pgmpy, libDAI and TensorInference.jl ran too, each on the programs it takes (its own paper's, the Dice programs, the ProbLog programs or the networks); a system's time is its process's, start-up included. Storm, PRISM, Ace, Merlin, pyAgrum, pgmpy, libDAI and TensorInference.jl ran in a later run, on other machines, on the instances the release run did not give them, each beside tddlog: their times compare with that run's tddlog, not with the other columns, and the comparisons below use it. Made by gen_roulette.py caesar 100.

;; benchmark/modular/caesar.rkt of the Roulette artifact (Zenodo
;; 10.5281/zenodo.15253254), its last line changed to print the posterior
;; of the key, key(K) for K = 0..25, in full (print.rkt), and to take a
;; second argument, `--no-complement`, which queries as `(query e #f)`.
;; By default `query` takes one value's probability as one minus the
;; others' (and the evidence's complement): once the evidence falls below
;; about 1e-16 that value is lost, as the key's most likely value is at
;; 30 letters. The artifact's hmm.rkt queries with #f for that reason.
(require "print.rkt")

(define rng (make-pseudo-random-generator))
(parameterize ([current-pseudo-random-generator rng]) (random-seed 0))

;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;

(require rosette/lib/destruct)

(define (natural->bv k size)
  (cond
    [(zero? size) '()]
    [else (cons (odd? k) (natural->bv (truncate (/ k 2)) (sub1 size)))]))

(define (bv->natural xs)
  (destruct xs
    [(list) 0]
    [(cons x xt) (+ (if x 1 0) (* 2 (bv->natural xt)))]))

(define (bvadd xs ys)
  (let go ([xs xs] [ys ys] [cin #f])
    (match* (xs ys)
      [('() '()) '()]
      [((cons x xt) (cons y yt))
       (define x⊕y (xor x y))
       (cons (xor x⊕y cin) (go xt yt (or (and x y) (and cin x⊕y))))])))

;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;

(require (only-in rosette assert))

(define (discrete . xs)
  (bin-cat (map cons (range (length xs)) xs)
           (inexact->exact (expt 2 (ceiling (log (length xs) 2))))))

(define (bin-cat xs size)
  (match xs
    [(list) (assert #f)]
    [(list (cons x _)) (natural->bv x size)]
    [_
     (define-values (left right)
       (split-at xs (floor (/ (length xs) 2))))
     (define left-sum (foldl + 0 (map cdr left)))
     (if (flip left-sum)
         (bin-cat (renormalize left left-sum) size)
         (bin-cat (renormalize right (- 1 left-sum)) size))]))

(define (renormalize xs n)
  (for/list ([x+y (in-list xs)])
    (cons (car x+y) (/ (cdr x+y) n))))

function report(k, x, d)
    if x isa AnyBool
        println("alea ", k, " true ", d[true])
        println("alea ", k, " false ", d[false])
    else
        for v in 0:(2^width(x) - 1)
            println("alea ", k, " ", v, " ", d[v])
        end
    end
end

function run_program()
    EV[] = true
    stats = @timed begin
        comps = components(dice_main())
        dists = pr(comps...; evidence = EV[])
        (comps, length(comps) == 1 ? [dists] : dists)
    end
    comps, dists = stats.value
    for k in eachindex(comps)
        report(k, comps[k], dists[k])
    end
    println("alea_seconds ", stats.time)
    println("alea_compile_seconds ", get(stats, :compile_time, NaN))
end

warmup()
task = Task(run_program, 1 << 30)
schedule(task)
wait(task)
