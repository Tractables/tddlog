rl_network_10

The scaling programs of the Roulette paper (Moy et al., PLDI 2025) on ladders of size, and its nine Bayesian networks. Roulette runs the artifact's programs; tddlog, Dice and Alea.jl run programs of the same models. On caesar Roulette's time is that of its query with #f, which keeps the answer. 600 s a run. Storm and PRISM ran too, each on the programs it takes (its own paper's, the Dice programs or the ProbLog programs); a system's time is its process's, start-up included. Storm and PRISM ran in a later run, on other machines, on the instances the release run did not give them, each beside tddlog: their times compare with that run's tddlog, not with the other columns, and the comparisons below use it. Made by gen_roulette.py network 10.

;; full (print.rkt): m(K,V) for a message whose bits, least significant
;; first, make V, m(K,16) for #f (lost). Roulette's distribution also holds #t
;; at a probability of 0 up to rounding; it is printed as other(#t).
(require (submod roulette/benchmark/scaling/network rsdd)
         roulette/example/private/pmf
         racket/format
         "print.rkt")
(define k (string->number (vector-ref (current-command-line-arguments) 0)))
(define result (main k))
(for ([(v p) (in-pmf result)])
  (cond
    [(not v) (emit (format "m(~a,16)" k) p)]
    [(list? v) (emit (format "m(~a,~a)" k (for/sum ([b v] [i (in-naturals)]) (if b (expt 2 i) 0))) p)]
    [else (emit (format "other(~a)" (~s v)) p)]))

fun diamond(m: (bool, (bool, (bool, (bool, bool))))) {
  let lost = fst m in
  let b0 = fst (snd m) in
  let b1 = fst (snd (snd m)) in
  let b2 = fst (snd (snd (snd m))) in
  let b3 = snd (snd (snd (snd m))) in
  let bit = discrete(0.25, 0.25, 0.25, 0.25) in
  let nz = flip 0.01 in
  let fw = flip 0.99 in
  let sp = flip 0.5 in
  let c = nz && !lost in
  let n0 = b0 ^ (c && bit == int(2, 0)) in
  let n1 = b1 ^ (c && bit == int(2, 1)) in
  let n2 = b2 ^ (c && bit == int(2, 2)) in
  let n3 = b3 ^ (c && bit == int(2, 3)) in
  if sp then (lost, (n0, (n1, (n2, n3))))
  else (if fw then m else (true, (false, (false, (false, false)))))
}

function warmup()
    a = flip(0.7)
    x = discrete(DistUInt{2}, [0.1, 0.2, 0.3, 0.4])
    y = uniform(DistUInt{3}, 1, 6)
    z = ifelse(a, x + DistUInt{2}(1), x - DistUInt{2}(1))

    pr(a, z, y, t...; evidence = e)
    nothing
end

function report(k, x, d)
    if x isa AnyBool
        println("alea ", k, " true ", d[true])
        println("alea ", k, " false ", d[false])
    else
        for v in 0:(2^width(x) - 1)
            println("alea ", k, " ", v, " ", d[v])
        end
    end
end

function run_program()
    EV[] = true
    stats = @timed begin
        comps = components(dice_main())
        dists = pr(comps...; evidence = EV[])
        (comps, length(comps) == 1 ? [dists] : dists)
    end
    comps, dists = stats.value
    for k in eachindex(comps)
        report(k, comps[k], dists[k])
    end
    println("alea_seconds ", stats.time)
    println("alea_compile_seconds ", get(stats, :compile_time, NaN))
end

warmup()
task = Task(run_program, 1 << 30)
schedule(task)
wait(task)
end
