    rain ~ Bernoulli(0.8);

    rain ~ Bernoulli(0.2);
    sprinkler ~ Bernoulli(0.5);
}

temp1 ~ Bernoulli(0.7);
if temp1 = 1 && rain = 1 {
    wetRoof := 1;
} else {
    wetRoof := 0;
}

temp3 ~ Bernoulli(0.9);

    wetGrass := 0;

observe(wetGrass = 1);

// grass.dice, translated by tddlog's scripts/dice2psi.py, component 0.
def main() {
  v1_cloudy := flip(0.5);
  v4_rain := 0;
  if v1_cloudy { v2_rain := flip(0.8); v4_rain = v2_rain; } else { v3_rain := flip(0.2); v4_rain = v3_rain; }
  v7_sprinkler := 0;
  if v1_cloudy { v5_sprinkler := flip(0.1); v7_sprinkler = v5_sprinkler; } else { v6_sprinkler := flip(0.5); v7_sprinkler = v6_sprinkler; }
  v8_temp1 := flip(0.7);
  v9_wetroof := (v8_temp1 && v4_rain);
  v10_temp2 := flip(0.9);
  v11_temp3 := flip(0.9);
  v12_wetgrass := (v10_temp2 && v4_rain);
  v13_wetgrass := (v11_temp3 && v7_sprinkler);
  v14_wetgrass := (v12_wetgrass || v13_wetgrass);
  observe(v14_wetgrass);
  return v4_rain;
}
