// alarm.dice, translated by tddlog's scripts/dice2psi.py, component 0.
def main() {
  v1_earthquake := flip(0.0001);
  v2_burglary := flip(0.001);
  v3_alarm := (v1_earthquake || v2_burglary);
  v6_phoneworking := 0;
  if v1_earthquake { v4_phoneworking := flip(0.7); v6_phoneworking = v4_phoneworking; } else { v5_phoneworking := flip(0.99); v6_phoneworking = v5_phoneworking; }
  v11_marywakes := 0;
  if v3_alarm { v9_marywakes := 0; if v1_earthquake { v7_marywakes := flip(0.8); v9_marywakes = v7_marywakes; } else { v8_marywakes := flip(0.6); v9_marywakes = v8_marywakes; } v11_marywakes = v9_marywakes; } else { v10_marywakes := flip(0.2); v11_marywakes = v10_marywakes; }
  v12_called := (v11_marywakes && v6_phoneworking);
  observe(v12_called);
  return v2_burglary;
}
