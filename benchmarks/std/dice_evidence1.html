// evidence1.dice, translated by tddlog's scripts/dice2psi.py, component 0.
def main() {
  v1_evidence := flip(0.5);
  v2_coin := flip(0.5);
  v3_result := 0;
  if v1_evidence { observe(v2_coin); v3_result = v1_evidence; } else {  v3_result = v1_evidence; }
  return v3_result;
}
