dice_caesar-iter_k1000

The programs and Bayesian networks of the Dice paper (Holtzen, Van den Broeck and Millstein, OOPSLA 2020), a network as one Dice program per queried variable, and its loop programs scaled up. tddlog runs them translated to ProbLog, Alea.jl translated to Julia, Roulette translated to Racket. 300 s a run (600 s on the loop programs). Geni, noDice, Psi, Genfer, SPPL, Pluck, PITA, aspmc, Storm, PRISM, Ace, Merlin, pyAgrum, pgmpy, libDAI and TensorInference.jl ran too, each on the programs it takes (its own paper's, the Dice programs, the ProbLog programs or the networks); a system's time is its process's, start-up included. ProbLog, Geni, Genfer, SPPL, Pluck, PITA, aspmc, Storm, PRISM, Ace, Merlin, pyAgrum, pgmpy, libDAI and TensorInference.jl ran in a later run, on other machines, on the instances the release run did not give them. Made by dice2problog.py dice_caesar-iter_k1000.dice.

Seconds, a time limit of 600 s; zero: every answer printed as 0 where the reference's are positive. skipped: not run: the system had failed on a smaller size of the same program. error: the system stopped with an error. >600: the time limit ran out. A process time is the whole process's, with the language's start and compilation; the other time of the same system leaves them out, as its paper times it. A time is the median of the repetitions below, a failed one counted as infinite. A system run in several forms (ProbLog's knowledge compilers, Ace's routes) has the time of its fastest form.

fun sendChar(arg: (int(2), int(2))) {
  let key = fst arg in
  let observation = snd arg in
  let gen = discrete(0.5, 0.25, 0.125, 0.125) in    // sample a FooLang character
  let enc = key + gen in                            // encrypt the character
  let tmp = observe observation == enc in
  (key, observation)
}

module AleaProgram
using Alea
# Julia would spend more time optimizing the long straight-line functions
# below than running them takes: they are interpreted, and Alea's own
# methods compile as usual.
Base.Experimental.@compiler_options compile=min optimize=0 infer=false

const EV = Ref{AnyBool}(true)
observe!(path, c) = (EV[] = EV[] & (!path | c); true)

function shr(x::DistUInt{W}, n::Integer) where W
    k = min(n, W)
    DistUInt{W}(vcat(fill(false, k), x.bits[1:W - k]))
end

components(x::Tuple) = reduce(vcat, map(components, x); init = Any[])
components(x) = Any[x]
width(::DistUInt{W}) where W = W

function warmup()
    a = flip(0.7)
    x = discrete(DistUInt{2}, [0.1, 0.2, 0.3, 0.4])
    y = uniform(DistUInt{3}, 1, 6)
    z = ifelse(a, x + DistUInt{2}(1), x - DistUInt{2}(1))
    t = ifelse(flip(0.4), (a, z), (!a, x * x))
    e = prob_equals(z, DistUInt{2}(2)) | (x < z) | xor(a, flip(0.2))
    pr(flip(0.7))
    pr(a, z, y, t...; evidence = e)
    nothing
end

function d_sendChar(x1_path, x2_arg)
    x3_key = x2_arg[1]
    x4_observation = x2_arg[2]
    x5_gen = discrete(DistUInt{2}, [0.5, 0.25, 0.125, 0.125])
    x6_enc = (x3_key + x5_gen)
    observe!(x1_path, prob_equals(x4_observation, x6_enc))
    x7_sendChar = (x3_key, x4_observation)
    return x7_sendChar
end

function dice_main()
    x8_key = discrete(DistUInt{2}, [0.25, 0.25, 0.25, 0.25])
    x9_tmp = (x8_key, DistUInt{2}(2))
    for _ in 1:1000
     x9_tmp = d_sendChar(true, x9_tmp)
    end
    return x8_key
end

function report(k, x, d)
    if x isa AnyBool
        println("alea ", k, " true ", d[true])
        println("alea ", k, " false ", d[false])
    else
        for v in 0:(2^width(x) - 1)
            println("alea ", k, " ", v, " ", d[v])
        end
    end
end

function run_program()
    EV[] = true
    stats = @timed begin
        comps = components(dice_main())
        dists = pr(comps...; evidence = EV[])
        (comps, length(comps) == 1 ? [dists] : dists)
    end
    comps, dists = stats.value
    for k in eachindex(comps)
        report(k, comps[k], dists[k])
    end
    println("alea_seconds ", stats.time)
    println("alea_compile_seconds ", get(stats, :compile_time, NaN))
end

warmup()
task = Task(run_program, 1 << 30)
schedule(task)
wait(task)
end

(define complement?
  (not (member "--no-complement" (vector->list (current-command-line-arguments)))))

;; Integers: lists of Booleans, least significant bit first.
(define (bxor a b) (! (<=> a b)))
(define (bv-add xs ys [c #f])
  (if (null? xs)
      '()
      (let* ([x (car xs)] [y (car ys)] [s (bxor x y)])
        (cons (bxor s c) (bv-add (cdr xs) (cdr ys) (|| (&& x y) (&& c s)))))))
(define (bv-sub xs ys) (bv-add xs (map ! ys) #t))
(define (bv-mul xs ys)
  (define w (length xs))
  (for/fold ([acc (make-list w #f)]) ([y (in-list ys)] [i (in-naturals)])
    (if (eq? y #f)
        acc
        (bv-add acc (append (make-list i #f) (for/list ([x (in-list (take xs (- w i)))]) (&& y x)))))))
(define (bv-eq xs ys) (apply && (map <=> xs ys)))
(define (bv-ult xs ys)
  (for/fold ([lt #f]) ([x (in-list xs)] [y (in-list ys)])
    (|| (&& (! x) y) (&& (<=> x y) lt))))
(define (bv-shl xs k)
  (define w (length xs))
  (take (append (make-list (min k w) #f) xs) w))
(define (bv-shr xs k)
  (define j (min k (length xs)))
  (append (drop xs j) (make-list j #f)))

;; Dice's categorical (gen_discrete): from the most significant bit down,
;; bit k is a flip whose probability is the mass of the values with bit k
;; set among those that agree on the bits above, over their mass; one flip
;; for each distinct ratio, chosen by the bits above, settings of no mass
;; left out.
(define (discrete w ps)
  (define size (expt 2 w))
  (define given (list->vector ps))
  (define masses
    (for/vector ([i (in-range size)])
      (if (< i (vector-length given)) (inexact->exact (vector-ref given i)) 0)))
  (define (mass lo hi) (for/sum ([i (in-range lo hi)]) (vector-ref masses i)))
  (let level ([k (sub1 w)] [above '()])
    (cond
      [(< k 0) (reverse above)]
      [else
       (define span (expt 2 (add1 k)))
       (define flips (make-hash))
       (define table
         (for/vector ([u (in-range (quotient size span))])
           (define lo (* u span))
           (define total (mass lo (+ lo span)))
           (cond
             [(zero? total) 'none]
             [else
              (define r (exact->inexact (/ (mass (+ lo (quotient span 2)) (+ lo span)) total)))
              (hash-ref! flips r (λ () (cond [(= r 0) #f] [(= r 1) #t] [else (flip r)])))])))
       (level (sub1 k) (append above (list (choose above table))))])))
;; The entry of table for the setting of the bits `above` (most significant
;; first), constant where every setting of mass agrees.
(define (choose above table)
  (define vals (for/list ([x (in-vector table)] #:unless (eq? x 'none)) x))
  (cond
    [(null? vals) #f]
    [(andmap (λ (x) (eq? x (car vals))) vals) (car vals)]
    [else
     (define half (quotient (vector-length table) 2))
     (if (car above)
         (choose (cdr above) (vector-copy table half))
         (choose (cdr above) (vector-copy table 0 half)))]))
(define (uniform w lo hi)
  (discrete w (for/list ([i (in-range hi)]) (if (< i lo) 0.0 (exact->inexact (/ 1 (- hi lo)))))))

(define (bits->natural bs)
  (for/sum ([b (in-list bs)] [i (in-naturals)]) (if b (expt 2 i) 0)))
(define (real p) (if (exact? p) (exact->inexact p) p))
;; Racket's `time` around the program's statements and its queries.
(define (start-clock)
  (vector (current-process-milliseconds) (current-inexact-milliseconds) (current-gc-milliseconds)))
(define (print-time c)
  (printf "cpu time: ~a real time: ~a gc time: ~a\n"
          (- (current-process-milliseconds) (vector-ref c 0))
          (inexact->exact (round (- (current-inexact-milliseconds) (vector-ref c 1))))
          (- (current-gc-milliseconds) (vector-ref c 2))))
(define (report-bool k pm)
  (printf "roulette ~a true ~a\n" k (real (pm #t)))
  (printf "roulette ~a false ~a\n" k (real (pm #f))))
(define (report-int k pm w)
  (define by-value (make-hash))
  (for ([(v p) (in-pmf pm)])
    (hash-set! by-value (bits->natural v) p))
  (for ([i (in-range (expt 2 w))])
    (printf "roulette ~a ~a ~a\n" k i (real (hash-ref by-value i 0)))))

def f_sendchar(v1_arg) {
  v2_key := v1_arg[0];
  v3_observation := v1_arg[1];
  v4_gen := categorical([1/2, 1/4, 1/8, 1/8]);
  v5_enc := ((v2_key + v4_gen) - 4 * ((v2_key + v4_gen) >= 4));
  v6_tmp := (v3_observation == v5_enc);
  observe(v6_tmp);
  v7_sendchar := (v2_key, v3_observation);
  return v7_sendchar;
}

def main() {
  v8_key := categorical([1/4, 1/4, 1/4, 1/4]);
  v9_tmp := (v8_key, 2);
  v10_state := v9_tmp;
  for _ in [0..1000) { v10_state = f_sendchar(v10_state); }
  return v8_key;
}
