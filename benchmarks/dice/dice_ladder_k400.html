dice_ladder_k400

The programs and Bayesian networks of the Dice paper (Holtzen, Van den Broeck and Millstein, OOPSLA 2020), a network as one Dice program per queried variable, and its loop programs scaled up. tddlog runs them translated to ProbLog, Alea.jl translated to Julia, Roulette translated to Racket. 300 s a run (600 s on the loop programs). Geni, noDice, Psi, Genfer, SPPL, Pluck, PITA, aspmc, Storm, PRISM, Ace, Merlin, pyAgrum, pgmpy, libDAI and TensorInference.jl ran too, each on the programs it takes (its own paper's, the Dice programs, the ProbLog programs or the networks); a system's time is its process's, start-up included. Geni, noDice, Psi, Genfer, SPPL, Pluck, PITA, aspmc, Storm, PRISM, Ace, Merlin, pyAgrum, pgmpy, libDAI and TensorInference.jl ran in a later run, on other machines, on the instances the release run did not give them, each beside tddlog: their times compare with that run's tddlog, not with the other columns, and the comparisons below use it. Made by dice2problog.py dice_ladder_k400.dice.

// The ladder network of the Dice paper's network verification benchmark
// (bench/dicebench.ml, gen_ladder), its repeated calls written as iterate.
fun ladder(s: (bool, bool)) {
  let s1 = fst s in
  let s2 = snd s in
  if s1 then
    if flip 0.001 then (false, false)
    else let f = flip 0.5 in (f, !f)
  else if s2 then
    let f = flip 0.5 in (f, !f)
  else (false, false)
}
iterate(ladder, (true, false), 400)

module AleaProgram
using Alea
# Julia would spend more time optimizing the long straight-line functions
# below than running them takes: they are interpreted, and Alea's own
# methods compile as usual.
Base.Experimental.@compiler_options compile=min optimize=0 infer=false

const EV = Ref{AnyBool}(true)
observe!(path, c) = (EV[] = EV[] & (!path | c); true)

function shr(x::DistUInt{W}, n::Integer) where W
    k = min(n, W)
    DistUInt{W}(vcat(fill(false, k), x.bits[1:W - k]))
end

components(x::Tuple) = reduce(vcat, map(components, x); init = Any[])
components(x) = Any[x]
width(::DistUInt{W}) where W = W

function warmup()
    a = flip(0.7)
    x = discrete(DistUInt{2}, [0.1, 0.2, 0.3, 0.4])
    y = uniform(DistUInt{3}, 1, 6)
    z = ifelse(a, x + DistUInt{2}(1), x - DistUInt{2}(1))
    t = ifelse(flip(0.4), (a, z), (!a, x * x))
    e = prob_equals(z, DistUInt{2}(2)) | (x < z) | xor(a, flip(0.2))
    pr(flip(0.7))
    pr(a, z, y, t...; evidence = e)
    nothing
end

function d_ladder(x1_path, x2_s)
    x3_s1 = x2_s[1]
    x4_s2 = x2_s[2]
    if x3_s1 !== false
     x5_ladder = flip(0.001)
     if x5_ladder !== false
      x6_ladder = (false, false)
     end
     if x5_ladder !== true
      x7_f = flip(0.5)
      x8_ladder = (x7_f, !(x7_f))
     end
     x9_ladder = x5_ladder === true ? x6_ladder : x5_ladder === false ? x8_ladder : ifelse(x5_ladder, x6_ladder, x8_ladder)
    end
    if x3_s1 !== true
     if x4_s2 !== false
      x10_f = flip(0.5)
      x11_ladder = (x10_f, !(x10_f))
     end
     if x4_s2 !== true
      x12_ladder = (false, false)
     end
     x13_ladder = x4_s2 === true ? x11_ladder : x4_s2 === false ? x12_ladder : ifelse(x4_s2, x11_ladder, x12_ladder)
    end
    x14_ladder = x3_s1 === true ? x9_ladder : x3_s1 === false ? x13_ladder : ifelse(x3_s1, x9_ladder, x13_ladder)
    return x14_ladder
end

function dice_main()
    x15_result = (true, false)
    for _ in 1:400
     x15_result = d_ladder(true, x15_result)
    end
    return x15_result
end

function report(k, x, d)
    if x isa AnyBool
        println("alea ", k, " true ", d[true])
        println("alea ", k, " false ", d[false])
    else
        for v in 0:(2^width(x) - 1)
            println("alea ", k, " ", v, " ", d[v])
        end
    end
end

function run_program()
    EV[] = true
    stats = @timed begin
        comps = components(dice_main())
        dists = pr(comps...; evidence = EV[])
        (comps, length(comps) == 1 ? [dists] : dists)
    end
    comps, dists = stats.value
    for k in eachindex(comps)
        report(k, comps[k], dists[k])
    end
    println("alea_seconds ", stats.time)
    println("alea_compile_seconds ", get(stats, :compile_time, NaN))
end

warmup()
task = Task(run_program, 1 << 30)
schedule(task)
wait(task)
end

  (not (member "--no-complement" (vector->list (current-command-line-arguments)))))

;; Integers: lists of Booleans, least significant bit first.
(define (bxor a b) (! (<=> a b)))
(define (bv-add xs ys [c #f])
  (if (null? xs)
      '()
      (let* ([x (car xs)] [y (car ys)] [s (bxor x y)])
        (cons (bxor s c) (bv-add (cdr xs) (cdr ys) (|| (&& x y) (&& c s)))))))
(define (bv-sub xs ys) (bv-add xs (map ! ys) #t))
(define (bv-mul xs ys)
  (define w (length xs))
  (for/fold ([acc (make-list w #f)]) ([y (in-list ys)] [i (in-naturals)])
    (if (eq? y #f)
        acc
        (bv-add acc (append (make-list i #f) (for/list ([x (in-list (take xs (- w i)))]) (&& y x)))))))
(define (bv-eq xs ys) (apply && (map <=> xs ys)))
(define (bv-ult xs ys)
  (for/fold ([lt #f]) ([x (in-list xs)] [y (in-list ys)])
    (|| (&& (! x) y) (&& (<=> x y) lt))))
(define (bv-shl xs k)
  (define w (length xs))
  (take (append (make-list (min k w) #f) xs) w))
(define (bv-shr xs k)

;; Dice's categorical (gen_discrete): from the most significant bit down,
;; bit k is a flip whose probability is the mass of the values with bit k
;; set among those that agree on the bits above, over their mass; one flip
;; for each distinct ratio, chosen by the bits above, settings of no mass
;; left out.
(define (discrete w ps)
  (define size (expt 2 w))
  (define given (list->vector ps))
  (define masses
    (for/vector ([i (in-range size)])
      (if (< i (vector-length given)) (inexact->exact (vector-ref given i)) 0)))
  (define (mass lo hi) (for/sum ([i (in-range lo hi)]) (vector-ref masses i)))
  (let level ([k (sub1 w)] [above '()])
    (cond
      [(< k 0) (reverse above)]
      [else
       (define span (expt 2 (add1 k)))
       (define flips (make-hash))
       (define table
         (for/vector ([u (in-range (quotient size span))])
           (define lo (* u span))
           (define total (mass lo (+ lo span)))
           (cond
             [(zero? total) 'none]
             [else
              (define r (exact->inexact (/ (mass (+ lo (quotient span 2)) (+ lo span)) total)))
              (hash-ref! flips r (λ () (cond [(= r 0) #f] [(= r 1) #t] [else (flip r)])))])))
       (level (sub1 k) (append above (list (choose above table))))])))
;; The entry of table for the setting of the bits `above` (most significant
;; first), constant where every setting of mass agrees.
(define (choose above table)
  (define vals (for/list ([x (in-vector table)] #:unless (eq? x 'none)) x))
  (cond
    [(null? vals) #f]
    [(andmap (λ (x) (eq? x (car vals))) vals) (car vals)]
    [else
     (define half (quotient (vector-length table) 2))
     (if (car above)
         (choose (cdr above) (vector-copy table half))
         (choose (cdr above) (vector-copy table 0 half)))]))
(define (uniform w lo hi)
  (discrete w (for/list ([i (in-range hi)]) (if (< i lo) 0.0 (exact->inexact (/ 1 (- hi lo)))))))

(define (bits->natural bs)
  (for/sum ([b (in-list bs)] [i (in-naturals)]) (if b (expt 2 i) 0)))
(define (real p) (if (exact? p) (exact->inexact p) p))
;; Racket's `time` around the program's statements and its queries.
(define (start-clock)

(define (report-bool k pm)
  (printf "roulette ~a true ~a\n" k (real (pm #t)))
  (printf "roulette ~a false ~a\n" k (real (pm #f))))
(define (report-int k pm w)

// dice_ladder_k400.dice, translated by tddlog's scripts/dice2psi.py, component 0.
def f_ladder(v1_s) {
  v2_s1 := v1_s[0];
  v3_s2 := v1_s[1];
  v15_ladder := 0;
  if v2_s1 { v4_ladder := flip(0.001); v9_ladder := 0; if v4_ladder { v5_ladder := (0, 0); v9_ladder = v5_ladder; } else { v6_f := flip(0.5); v7_ladder := !v6_f; v8_ladder := (v6_f, v7_ladder); v9_ladder = v8_ladder; } v15_ladder = v9_ladder; } else { v14_ladder := 0; if v3_s2 { v10_f := flip(0.5); v11_ladder := !v10_f; v12_ladder := (v10_f, v11_ladder); v14_ladder = v12_ladder; } else { v13_ladder := (0, 0); v14_ladder = v13_ladder; } v15_ladder = v14_ladder; }
  return v15_ladder;
}

def main() {
  v16_result := (1, 0);
  v17_state := v16_result;
  for _ in [0..400) { v17_state = f_ladder(v17_state); }
  return v17_state[0];
}
