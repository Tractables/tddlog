bn_cancer

The programs and Bayesian networks of the Dice paper (Holtzen, Van den Broeck and Millstein, OOPSLA 2020), a network as one Dice program per queried variable, and its loop programs scaled up. tddlog runs them translated to ProbLog, Alea.jl translated to Julia, Roulette translated to Racket. 300 s a run (600 s on the loop programs). Geni, noDice, Psi, Genfer, SPPL, Pluck, PITA, aspmc, Storm, PRISM, Ace, Merlin, pyAgrum, pgmpy, libDAI and TensorInference.jl ran too, each on the programs it takes (its own paper's, the Dice programs, the ProbLog programs or the networks); a system's time is its process's, start-up included. ProbLog, Geni, Genfer, SPPL, Pluck, PITA, aspmc, Storm, PRISM, Ace, Merlin, pyAgrum, pgmpy, libDAI and TensorInference.jl ran in a later run, on other machines, on the instances the release run did not give them. using Alea
# Julia would spend more time optimizing the long straight-line functions
# below than running them takes: they are interpreted, and Alea's own
# methods compile as usual.
Base.Experimental.@compiler_options compile=min optimize=0 infer=false

const EV = Ref{AnyBool}(true)
observe!(path, c) = (EV[] = EV[] & (!path | c); true)

function shr(x::DistUInt{W}, n::Integer) where W
    k = min(n, W)
    DistUInt{W}(vcat(fill(false, k), x.bits[1:W - k]))
end

components(x::Tuple) = reduce(vcat, map(components, x); init = Any[])
components(x) = Any[x]
width(::DistUInt{W}) where W = W

function warmup()
    a = flip(0.7)
    x = discrete(DistUInt{2}, [0.1, 0.2, 0.3, 0.4])
    y = uniform(DistUInt{3}, 1, 6)
    z = ifelse(a, x + DistUInt{2}(1), x - DistUInt{2}(1))
    t = ifelse(flip(0.4), (a, z), (!a, x * x))
    e = prob_equals(z, DistUInt{2}(2)) | (x < z) | xor(a, flip(0.2))
    pr(flip(0.7))
    pr(a, z, y, t...; evidence = e)
    nothing
end

function dice_main()
    x1_POLLUTION = discrete(DistUInt{1}, [0.9, 0.1])
    x2_SMOKER = discrete(DistUInt{1}, [0.3, 0.7])
    x3_CANCER = prob_equals(x1_POLLUTION, DistUInt{1}(0))
    if x3_CANCER !== false
     x4_CANCER = prob_equals(x2_SMOKER, DistUInt{1}(0))
     if x4_CANCER !== false
      x5_CANCER = discrete(DistUInt{1}, [0.03, 0.97])
     end
     if x4_CANCER !== true
      x6_CANCER = discrete(DistUInt{1}, [0.001, 0.999])
     end
     x7_CANCER = x4_CANCER === true ? x5_CANCER : x4_CANCER === false ? x6_CANCER : ifelse(x4_CANCER, x5_CANCER, x6_CANCER)
    end
    if x3_CANCER !== true
     x8_CANCER = prob_equals(x2_SMOKER, DistUInt{1}(0))
     if x8_CANCER !== false
      x9_CANCER = discrete(DistUInt{1}, [0.05, 0.95])
     end
     if x8_CANCER !== true
      x10_CANCER = discrete(DistUInt{1}, [0.02, 0.98])
     end
     x11_CANCER = x8_CANCER === true ? x9_CANCER : x8_CANCER === false ? x10_CANCER : ifelse(x8_CANCER, x9_CANCER, x10_CANCER)
    end
    x12_CANCER = x3_CANCER === true ? x7_CANCER : x3_CANCER === false ? x11_CANCER : ifelse(x3_CANCER, x7_CANCER, x11_CANCER)
    x13_XRAY = prob_equals(x12_CANCER, DistUInt{1}(0))
    if x13_XRAY !== false
     x14_XRAY = discrete(DistUInt{1}, [0.9, 0.1])
    end
    if x13_XRAY !== true
     x15_XRAY = discrete(DistUInt{1}, [0.2, 0.8])
    end
    x16_XRAY = x13_XRAY === true ? x14_XRAY : x13_XRAY === false ? x15_XRAY : ifelse(x13_XRAY, x14_XRAY, x15_XRAY)
    x17_DYSPNOEA = prob_equals(x12_CANCER, DistUInt{1}(0))
    if x17_DYSPNOEA !== false
     x18_DYSPNOEA = discrete(DistUInt{1}, [0.65, 0.35])
    end
    if x17_DYSPNOEA !== true
     x19_DYSPNOEA = discrete(DistUInt{1}, [0.3, 0.7])
    end

    x21_result = (x16_XRAY, x20_DYSPNOEA)
    return x21_result
end

function report(k, x, d)
    if x isa AnyBool
        println("alea ", k, " true ", d[true])
        println("alea ", k, " false ", d[false])
    else
        for v in 0:(2^width(x) - 1)
            println("alea ", k, " ", v, " ", d[v])
        end
    end
end

function run_program()
    EV[] = true
    stats = @timed begin
        comps = components(dice_main())
        dists = pr(comps...; evidence = EV[])
        (comps, length(comps) == 1 ? [dists] : dists)
    end
    comps, dists = stats.value
    for k in eachindex(comps)
        report(k, comps[k], dists[k])
    end
    println("alea_seconds ", stats.time)
    println("alea_compile_seconds ", get(stats, :compile_time, NaN))
end

warmup()
task = Task(run_program, 1 << 30)
schedule(task)
wait(task)
end

(define complement?
  (not (member "--no-complement" (vector->list (current-command-line-arguments)))))

;; Integers: lists of Booleans, least significant bit first.
(define (bxor a b) (! (<=> a b)))
(define (bv-add xs ys [c #f])
  (if (null? xs)
      '()
      (let* ([x (car xs)] [y (car ys)] [s (bxor x y)])
        (cons (bxor s c) (bv-add (cdr xs) (cdr ys) (|| (&& x y) (&& c s)))))))
(define (bv-sub xs ys) (bv-add xs (map ! ys) #t))
(define (bv-mul xs ys)
  (define w (length xs))
  (for/fold ([acc (make-list w #f)]) ([y (in-list ys)] [i (in-naturals)])
    (if (eq? y #f)
        acc
        (bv-add acc (append (make-list i #f) (for/list ([x (in-list (take xs (- w i)))]) (&& y x)))))))
(define (bv-eq xs ys) (apply && (map <=> xs ys)))
(define (bv-ult xs ys)
  (for/fold ([lt #f]) ([x (in-list xs)] [y (in-list ys)])
    (|| (&& (! x) y) (&& (<=> x y) lt))))
(define (bv-shl xs k)
  (define w (length xs))
  (take (append (make-list (min k w) #f) xs) w))
(define (bv-shr xs k)
  (define j (min k (length xs)))
  (append (drop xs j) (make-list j #f)))

;; Dice's categorical (gen_discrete): from the most significant bit down,
;; bit k is a flip whose probability is the mass of the values with bit k
;; set among those that agree on the bits above, over their mass; one flip
;; for each distinct ratio, chosen by the bits above, settings of no mass
;; left out.
(define (discrete w ps)
  (define size (expt 2 w))
  (define given (list->vector ps))
  (define masses
    (for/vector ([i (in-range size)])
      (if (< i (vector-length given)) (inexact->exact (vector-ref given i)) 0)))
  (define (mass lo hi) (for/sum ([i (in-range lo hi)]) (vector-ref masses i)))
  (let level ([k (sub1 w)] [above '()])
    (cond
      [(< k 0) (reverse above)]
      [else
       (define span (expt 2 (add1 k)))
       (define flips (make-hash))
       (define table
         (for/vector ([u (in-range (quotient size span))])
           (define lo (* u span))
           (define total (mass lo (+ lo span)))
           (cond
             [(zero? total) 'none]
             [else
              (define r (exact->inexact (/ (mass (+ lo (quotient span 2)) (+ lo span)) total)))
              (hash-ref! flips r (λ () (cond [(= r 0) #f] [(= r 1) #t] [else (flip r)])))])))
       (level (sub1 k) (append above (list (choose above table))))])))
;; The entry of table for the setting of the bits `above` (most significant
;; first), constant where every setting of mass agrees.
(define (choose above table)
  (define vals (for/list ([x (in-vector table)] #:unless (eq? x 'none)) x))
  (cond
    [(null? vals) #f]
    [(andmap (λ (x) (eq? x (car vals))) vals) (car vals)]
    [else
     (define half (quotient (vector-length table) 2))
     (if (car above)
         (choose (cdr above) (vector-copy table half))
         (choose (cdr above) (vector-copy table 0 half)))]))
(define (uniform w lo hi)
  (discrete w (for/list ([i (in-range hi)]) (if (< i lo) 0.0 (exact->inexact (/ 1 (- hi lo)))))))

(define (bits->natural bs)
  (for/sum ([b (in-list bs)] [i (in-naturals)]) (if b (expt 2 i) 0)))
(define (real p) (if (exact? p) (exact->inexact p) p))
;; Racket's `time` around the program's statements and its queries.
(define (start-clock)

(define (print-time c)

          (inexact->exact (round (- (current-inexact-milliseconds) (vector-ref c 1))))

(define (report-bool k pm)
  (printf "roulette ~a true ~a\n" k (real (pm #t)))
  (printf "roulette ~a false ~a\n" k (real (pm #f))))
