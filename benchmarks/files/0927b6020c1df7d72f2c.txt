function shr(x::DistUInt{W}, n::Integer) where W
    k = min(n, W)
    DistUInt{W}(vcat(fill(false, k), x.bits[1:W - k]))
end

components(x::Tuple) = reduce(vcat, map(components, x); init = Any[])

width(::DistUInt{W}) where W = W

function warmup()
    a = flip(0.7)
    x = discrete(DistUInt{2}, [0.1, 0.2, 0.3, 0.4])
    y = uniform(DistUInt{3}, 1, 6)
    z = ifelse(a, x + DistUInt{2}(1), x - DistUInt{2}(1))
    t = ifelse(flip(0.4), (a, z), (!a, x * x))
    e = prob_equals(z, DistUInt{2}(2)) | (x < z) | xor(a, flip(0.2))
    pr(flip(0.7))
    pr(a, z, y, t...; evidence = e)
    nothing
end

function d_diamond(x1_path, x2_m)
    x3_lost = x2_m[1]
    x4_b0 = x2_m[2][1]
    x5_b1 = x2_m[2][2][1]
    x6_b2 = x2_m[2][2][2][1]
    x7_b3 = x2_m[2][2][2][2]
    x8_bit = discrete(DistUInt{2}, [0.25, 0.25, 0.25, 0.25])
    x9_nz = flip(0.01)
    x10_fw = flip(0.99)
    x11_sp = flip(0.5)
    x12_c = (x9_nz & !(x3_lost))
    x13_n0 = xor(x4_b0, (x12_c & prob_equals(x8_bit, DistUInt{2}(0))))
    x14_n1 = xor(x5_b1, (x12_c & prob_equals(x8_bit, DistUInt{2}(1))))
    x15_n2 = xor(x6_b2, (x12_c & prob_equals(x8_bit, DistUInt{2}(2))))
    x16_n3 = xor(x7_b3, (x12_c & prob_equals(x8_bit, DistUInt{2}(3))))
    if x11_sp !== false
     x17_diamond = (x3_lost, (x13_n0, (x14_n1, (x15_n2, x16_n3))))
    end
    if x11_sp !== true
     if x10_fw !== true
      x18_diamond = (true, (false, (false, (false, false))))
     end
     x19_diamond = x10_fw === true ? x2_m : x10_fw === false ? x18_diamond : ifelse(x10_fw, x2_m, x18_diamond)
    end
    x20_diamond = x11_sp === true ? x17_diamond : x11_sp === false ? x19_diamond : ifelse(x11_sp, x17_diamond, x19_diamond)
    return x20_diamond
end

function report(k, x, d)
    if x isa AnyBool
        println("alea ", k, " true ", d[true])
        println("alea ", k, " false ", d[false])
    else
        for v in 0:(2^width(x) - 1)
            println("alea ", k, " ", v, " ", d[v])
        end
    end
end

function run_program()
    EV[] = true
    stats = @timed begin
        comps = components(dice_main())
        dists = pr(comps...; evidence = EV[])
        (comps, length(comps) == 1 ? [dists] : dists)
    end
    comps, dists = stats.value
    for k in eachindex(comps)
        report(k, comps[k], dists[k])
    end
    println("alea_seconds ", stats.time)
    println("alea_compile_seconds ", get(stats, :compile_time, NaN))
end

warmup()
