    x2647__goal = x2645__goal === true ? x2646__goal : x2645__goal === false ? x2641__goal : ifelse(x2645__goal, x2646__goal, x2641__goal)
    x2648_res = !(x2647__goal)
    if x2648_res !== false
     x2649_res = d_step(true, x2644_res)
    end
    x2650_res = x2648_res === true ? x2649_res : x2648_res === false ? x2644_res : ifelse(x2648_res, x2649_res, x2644_res)
    x2651_result = d_hit(true, x2650_res)
    x2652_result = (x2651_result, false)
    return x2652_result
end

function dice_main()
    (x646_res, x649__goal, x650_res,) = chunk_1()
    (x1045__goal, x1048_res, x1049__goal,) = chunk_2(x646_res, x649__goal, x650_res)
    (x1444_res, x1447__goal, x1448_res,) = chunk_3(x1045__goal, x1048_res, x1049__goal)
    (x1843__goal, x1846_res, x1847__goal,) = chunk_4(x1444_res, x1447__goal, x1448_res)
    (x2242_res, x2245__goal, x2246_res,) = chunk_5(x1843__goal, x1846_res, x1847__goal)
    (x2641__goal, x2644_res, x2645__goal,) = chunk_6(x2242_res, x2245__goal, x2246_res)
    x2652_result = chunk_7(x2641__goal, x2644_res, x2645__goal)
    return x2652_result
end

function report(k, x, d)
    if x isa AnyBool
        println("alea ", k, " true ", d[true])
        println("alea ", k, " false ", d[false])
    else
        for v in 0:(2^width(x) - 1)
            println("alea ", k, " ", v, " ", d[v])
        end
    end
end

function run_program()
    EV[] = true
    stats = @timed begin
        comps = components(dice_main())
        dists = pr(comps...; evidence = EV[])
        (comps, length(comps) == 1 ? [dists] : dists)
    end
    comps, dists = stats.value
    for k in eachindex(comps)
        report(k, comps[k], dists[k])
    end
    println("alea_seconds ", stats.time)
    println("alea_compile_seconds ", get(stats, :compile_time, NaN))
end

warmup()
task = Task(run_program, 1 << 30)
schedule(task)
wait(task)
end
