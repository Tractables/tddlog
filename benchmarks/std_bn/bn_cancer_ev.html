In the repository: examples/cancer.net.

% The lung cancer network of K. B. Korb and A. E. Nicholson, Bayesian
% Artificial Intelligence, Chapman and Hall/CRC, 2004 (second edition
% 2010), Section 2.2, written out from the book's description: pollution
% and smoking cause cancer, which a positive X-ray and dyspnoea follow.
% Hugin format, as scripts/gen_bn.py reads it.
net
{
}
node Pollution
{
  states = ( "low" "high" );
}
node Smoker
{
  states = ( "True" "False" );
}
node Cancer
{
  states = ( "True" "False" );
}
node Xray
{
  states = ( "positive" "negative" );
}
node Dyspnoea
{
  states = ( "True" "False" );
}
potential ( Pollution )
{
  data = ( 0.9 0.1 );
}
potential ( Smoker )
{
  data = ( 0.3 0.7 );
}
potential ( Cancer | Pollution Smoker )
{
  data = ((( 0.03 0.97 )     % Pollution = low, Smoker = True
           ( 0.001 0.999 ))  % Pollution = low, Smoker = False
          (( 0.05 0.95 )     % Pollution = high, Smoker = True
           ( 0.02 0.98 )));  % Pollution = high, Smoker = False
}
potential ( Xray | Cancer )
{
  data = (( 0.9 0.1 )        % Cancer = True
          ( 0.2 0.8 ));      % Cancer = False
}
potential ( Dyspnoea | Cancer )
{
  data = (( 0.65 0.35 )      % Cancer = True
          ( 0.3 0.7 ));      % Cancer = False
}

:- pita.
:- begin_lpad.
pollution(low):0.9 ; pollution(high):0.1.
smoker(true):0.3 ; smoker(false):0.7.
cancer(true):0.03 ; cancer(false):0.97 :- pollution(low), smoker(true).
cancer(true):0.001 ; cancer(false):0.999 :- pollution(low), smoker(false).
cancer(true):0.05 ; cancer(false):0.95 :- pollution(high), smoker(true).
cancer(true):0.02 ; cancer(false):0.98 :- pollution(high), smoker(false).
xray(positive):0.9 ; xray(negative):0.1 :- cancer(true).
xray(positive):0.2 ; xray(negative):0.8 :- cancer(false).
dyspnoea(true):0.65 ; dyspnoea(false):0.35 :- cancer(true).
dyspnoea(true):0.3 ; dyspnoea(false):0.7 :- cancer(false).
:- end_lpad.
main :-
    forall((Q = dyspnoea(true)), (prob(Q, (xray(negative)), P), format("~q:\t~w~n", [Q, P]))),
    forall((Q = dyspnoea(false)), (prob(Q, (xray(negative)), P), format("~q:\t~w~n", [Q, P]))).
