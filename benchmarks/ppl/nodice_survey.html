nodice_survey

Cao et al.'s integer arithmetic programs (UAI 2023), Pluck's sequence models (PLDI 2025) and noDice's purely probabilistic programs (OOPSLA 2026). Alea.jl runs an Alea.jl program itself; tddlog and Dice run the Boolean circuit Alea.jl builds for it, and Roulette that circuit's Dice program translated to Racket. The noDice programs are Dice programs, translated, and Roulette runs them translated to Racket. 600 s a run. Geni, noDice, Psi, Genfer, SPPL, Pluck, PITA, aspmc, Storm, PRISM, Ace, Merlin, pyAgrum, pgmpy, libDAI and TensorInference.jl ran too, each on the programs it takes (its own paper's, the Dice programs, the ProbLog programs or the networks); a system's time is its process's, start-up included. Roulette, Roulette's process, Geni, noDice, Psi, Genfer, SPPL, Pluck, PITA, aspmc, Storm, PRISM, Ace, Merlin, pyAgrum, pgmpy, libDAI and TensorInference.jl ran in a later run, on other machines, on the instances the release run did not give them. # below than running them takes: they are interpreted, and Alea's own

Base.Experimental.@compiler_options compile=min optimize=0 infer=false

observe!(path, c) = (EV[] = EV[] & (!path | c); true)

function shr(x::DistUInt{W}, n::Integer) where W
    k = min(n, W)
    DistUInt{W}(vcat(fill(false, k), x.bits[1:W - k]))
end

components(x::Tuple) = reduce(vcat, map(components, x); init = Any[])
components(x) = Any[x]
width(::DistUInt{W}) where W = W

function warmup()
    a = flip(0.7)
    x = discrete(DistUInt{2}, [0.1, 0.2, 0.3, 0.4])
    y = uniform(DistUInt{3}, 1, 6)
    z = ifelse(a, x + DistUInt{2}(1), x - DistUInt{2}(1))
    t = ifelse(flip(0.4), (a, z), (!a, x * x))
    e = prob_equals(z, DistUInt{2}(2)) | (x < z) | xor(a, flip(0.2))
    pr(flip(0.7))
    pr(a, z, y, t...; evidence = e)
    nothing
end

function report(k, x, d)
    if x isa AnyBool
        println("alea ", k, " true ", d[true])
        println("alea ", k, " false ", d[false])
    else
        for v in 0:(2^width(x) - 1)
            println("alea ", k, " ", v, " ", d[v])
        end
    end
end

function run_program()
    EV[] = true
    stats = @timed begin
        comps = components(dice_main())
        dists = pr(comps...; evidence = EV[])
        (comps, length(comps) == 1 ? [dists] : dists)
    end
    comps, dists = stats.value
    for k in eachindex(comps)
        report(k, comps[k], dists[k])
    end
    println("alea_seconds ", stats.time)
    println("alea_compile_seconds ", get(stats, :compile_time, NaN))
end

warmup()

schedule(task)
wait(task)

(define (bxor a b) (! (<=> a b)))
(define (bv-add xs ys [c #f])
  (if (null? xs)
      '()
      (let* ([x (car xs)] [y (car ys)] [s (bxor x y)])
        (cons (bxor s c) (bv-add (cdr xs) (cdr ys) (|| (&& x y) (&& c s)))))))
(define (bv-sub xs ys) (bv-add xs (map ! ys) #t))
(define (bv-mul xs ys)
  (define w (length xs))
  (for/fold ([acc (make-list w #f)]) ([y (in-list ys)] [i (in-naturals)])
    (if (eq? y #f)
        acc
        (bv-add acc (append (make-list i #f) (for/list ([x (in-list (take xs (- w i)))]) (&& y x)))))))
(define (bv-eq xs ys) (apply && (map <=> xs ys)))
(define (bv-ult xs ys)
  (for/fold ([lt #f]) ([x (in-list xs)] [y (in-list ys)])
    (|| (&& (! x) y) (&& (<=> x y) lt))))
(define (bv-shl xs k)
  (define w (length xs))
  (take (append (make-list (min k w) #f) xs) w))
(define (bv-shr xs k)
  (define j (min k (length xs)))
  (append (drop xs j) (make-list j #f)))

;; Dice's categorical (gen_discrete): from the most significant bit down,
;; bit k is a flip whose probability is the mass of the values with bit k
;; set among those that agree on the bits above, over their mass; one flip
;; for each distinct ratio, chosen by the bits above, settings of no mass
;; left out.
(define (discrete w ps)
  (define size (expt 2 w))
  (define given (list->vector ps))
  (define masses
    (for/vector ([i (in-range size)])
      (if (< i (vector-length given)) (inexact->exact (vector-ref given i)) 0)))
  (define (mass lo hi) (for/sum ([i (in-range lo hi)]) (vector-ref masses i)))
  (let level ([k (sub1 w)] [above '()])
    (cond
      [(< k 0) (reverse above)]
      [else
       (define span (expt 2 (add1 k)))
       (define flips (make-hash))
       (define table
         (for/vector ([u (in-range (quotient size span))])
           (define lo (* u span))
           (define total (mass lo (+ lo span)))
           (cond
             [(zero? total) 'none]
             [else
              (define r (exact->inexact (/ (mass (+ lo (quotient span 2)) (+ lo span)) total)))
              (hash-ref! flips r (λ () (cond [(= r 0) #f] [(= r 1) #t] [else (flip r)])))])))
       (level (sub1 k) (append above (list (choose above table))))])))
;; The entry of table for the setting of the bits `above` (most significant
;; first), constant where every setting of mass agrees.
(define (choose above table)
  (define vals (for/list ([x (in-vector table)] #:unless (eq? x 'none)) x))
  (cond
    [(null? vals) #f]
    [(andmap (λ (x) (eq? x (car vals))) vals) (car vals)]
    [else
     (define half (quotient (vector-length table) 2))
     (if (car above)
         (choose (cdr above) (vector-copy table half))
         (choose (cdr above) (vector-copy table 0 half)))]))
(define (uniform w lo hi)
  (discrete w (for/list ([i (in-range hi)]) (if (< i lo) 0.0 (exact->inexact (/ 1 (- hi lo)))))))

(define (bits->natural bs)

(define (real p) (if (exact? p) (exact->inexact p) p))
